# error-characteristics: one sensor error at a time, without a physics engine (README.md).
#
#   cmake -S tools/error-characteristics -B build-errors -DCMAKE_BUILD_TYPE=Release \
#     -DERROR_CHARACTERISTICS_IMPLEMENTATIONS=hakoniwa \
#     -DERROR_CHARACTERISTICS_LIBRARIES="<the libraries of those implementations>"
#
# For the Hakoniwa implementation, the libraries are those of a hakoniwa-sensor-pro release archive:
#   lib/libhakoniwa_mujoco_sensors_adapters.a  (the "hakoniwa" entry point)
#   lib/libhakoniwa_imu_gnss_sensor_adapters.a lib/libhakoniwa_imu_gnss_sensor_models.a
cmake_minimum_required(VERSION 3.16)
project(error_characteristics CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

get_filename_component(_adapter_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(ERROR_CHARACTERISTICS_IMPLEMENTATIONS "hakoniwa" CACHE STRING
    "Linked sensor implementations by name, for example hakoniwa;gazebo")
set(ERROR_CHARACTERISTICS_LIBRARIES "" CACHE STRING "Libraries of the listed implementations")

# The lookup of each implementation by name, generated from the names (sensor_adapter_implementation.hpp).
set(_declarations "")
set(_lookups "")
foreach(_name IN LISTS ERROR_CHARACTERISTICS_IMPLEMENTATIONS)
    if(NOT _name MATCHES "^[A-Za-z_][A-Za-z0-9_]*$")
        message(FATAL_ERROR "implementation name is not a C identifier: ${_name}")
    endif()
    string(APPEND _declarations "HAKO_SENSOR_ADAPTER_DECLARE_IMPLEMENTATION(${_name});\n")
    string(APPEND _lookups
        "    if (name == \"${_name}\") return HAKO_SENSOR_ADAPTER_IMPLEMENTATION_ENTRY(${_name})();\n")
endforeach()
set(_lookup_source "${CMAKE_CURRENT_BINARY_DIR}/generated/implementations.cpp")
file(WRITE "${_lookup_source}.in"
"// Generated by CMake from ERROR_CHARACTERISTICS_IMPLEMENTATIONS. Do not edit.
#include \"hakoniwa/sensor_adapter/sensor_adapter_implementation.hpp\"
#include <string>

${_declarations}
const hakoniwa::sensor_adapter::SensorAdapterImplementation* error_characteristics_find_implementation(
    const std::string& name)
{
${_lookups}    return nullptr;
}
")
configure_file("${_lookup_source}.in" "${_lookup_source}" COPYONLY)

add_executable(error-characteristics
    error_characteristics.cpp
    "${_lookup_source}"
    "${_adapter_root}/src/host/sensor_host.cpp")
target_include_directories(error-characteristics PRIVATE "${_adapter_root}/include")
target_link_libraries(error-characteristics PRIVATE ${ERROR_CHARACTERISTICS_LIBRARIES})
